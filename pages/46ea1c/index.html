并发任务控制

# demo

class TaskManager {
	constructor(paralleCount) {
		this.taskList = [];
		this.paralleCount = paralleCount;
		this.runingCount = 0;
	}
	add(task) {
		return new Promise((resolve, reject) => {
			this.taskList.push({
				task,
				resolve,
				reject,
			});
			this._run();
		});
	}

	_run() {
		while (this.runingCount < this.paralleCount && this.taskList.length) {
			const { task, resolve, reject } = this.taskList.shift();
			this.runingCount++;
			task()
				.then(resolve, reject)
				.finally(() => {
					this.runingCount--;
					this._run();
				});
		}
	}
}

function timeout(time) {
	return new Promise((resolve) => {
		setTimeout(() => {
			resolve();
		}, time);
	});
}

const taskManager = new TaskManager(2);
function addTask(time, name) {
	taskManager.add(() => timeout(time)).then((res) => console.log(`${name}完成`));
}

addTask(10000, 1);
addTask(1000, 2);
addTask(3000, 3);
addTask(1000, 4);
addTask(8000, 5);
// 输出23415
